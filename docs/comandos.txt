COMANDOS DOCKER (RESUMO)

cd "C:\Users\Jose\Desktop\Projeto de estagio"

# Subir só frontend (modo recomendado)
docker compose up -d --build web

# Subir frontend + backend + base de dados (os 3)
docker compose --profile docker-api --profile bundled-db up -d --build

# Subir frontend + backend (sem base de dados em Docker)
docker compose --profile docker-api up -d --build

# Subir frontend + base de dados (sem backend em Docker)
docker compose --profile bundled-db up -d --build

# Ver estado
docker ps
docker compose ps

# Ver logs
docker compose logs -f
docker compose logs -f frontend
docker compose logs -f backend
docker compose logs -f base-de-dados

# Parar
docker compose stop
docker compose stop frontend
docker compose stop backend
docker compose stop base-de-dados

# Desligar e remover containers/rede deste projeto
docker compose down

# Limpeza de containers antigos (se existirem)
docker rm -f frontend backend base-de-dados 2>$null
